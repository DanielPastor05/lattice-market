cmake_minimum_required(VERSION 3.20)
project(lattice_market VERSION 0.2.0 LANGUAGES CXX)
option(LATTICE_BENCH_TESTS "Run pinned DuckDB differential checks" OFF)
option(LATTICE_ASAN "Instrument the engine with AddressSanitizer" OFF)
option(LATTICE_FUZZ "Build a bounded Clang libFuzzer target" OFF)
add_executable(lattice src/main.cpp)
if(WIN32)
  add_executable(memory_probe tools/memory_probe.cpp)
  target_compile_features(memory_probe PRIVATE cxx_std_17)
  target_link_libraries(memory_probe PRIVATE psapi)
endif()
target_compile_features(lattice PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(lattice PRIVATE /W4 /permissive- /EHsc)
else()
  target_compile_options(lattice PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(LATTICE_ASAN)
  if(MSVC)
    string(REPLACE "/RTC1" "" CMAKE_CXX_FLAGS_DEBUG "${CMAKE_CXX_FLAGS_DEBUG}")
    target_compile_options(lattice PRIVATE /fsanitize=address /Zi)
    target_link_options(lattice PRIVATE /INCREMENTAL:NO)
    get_filename_component(LATTICE_COMPILER_BIN "${CMAKE_CXX_COMPILER}" DIRECTORY)
    find_file(LATTICE_ASAN_RUNTIME clang_rt.asan_dynamic-x86_64.dll
      PATHS "${LATTICE_COMPILER_BIN}" NO_DEFAULT_PATH REQUIRED)
    add_custom_command(TARGET lattice POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${LATTICE_ASAN_RUNTIME}" "$<TARGET_FILE_DIR:lattice>")
  else()
    target_compile_options(lattice PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(lattice PRIVATE -fsanitize=address,undefined)
  endif()
endif()
include(CTest)
if(LATTICE_FUZZ)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR WIN32)
    message(FATAL_ERROR "LATTICE_FUZZ requires Clang on Linux/macOS")
  endif()
  add_executable(lattice_fuzz tests/fuzz_reader.cpp)
  target_compile_features(lattice_fuzz PRIVATE cxx_std_20)
  target_compile_options(lattice_fuzz PRIVATE -fsanitize=fuzzer,address,undefined -fno-omit-frame-pointer)
  target_link_options(lattice_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
endif()
if(BUILD_TESTING)
  find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_engine.py $<TARGET_FILE:lattice>)
  add_test(NAME reader_mutations COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tools/fuzz_reader.py --lattice $<TARGET_FILE:lattice> --cases 24)
  if(LATTICE_BENCH_TESTS)
    add_test(NAME duckdb_differential COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_duckdb.py $<TARGET_FILE:lattice>)
  endif()
endif()
